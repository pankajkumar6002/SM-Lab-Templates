%% ---------------------------------------------------------------------
%%  1. DOCUMENT CLASS
%% ---------------------------------------------------------------------
%%  aspectratio=169 -> widescreen (16:9, modern projectors)
%%  aspectratio=43  -> old square-ish screens (4:3)
%%  11pt            -> base font size (options: 8pt 9pt 10pt 11pt 12pt)
%%  Add ",handout" to make a version with no animations (for printing).
\documentclass[aspectratio=169, 10pt]{beamer}

%% ---------------------------------------------------------------------
%%  2. PACKAGES (tools that add features)
%% ---------------------------------------------------------------------
\usepackage[T1]{fontenc}     % better font encoding
\usepackage{lmodern}         % clean, scalable fonts
\usepackage{graphicx}        % for images
\usepackage{booktabs}        % nice-looking tables
\usepackage{amsmath}         % math formulas
\usepackage{listings}        % code blocks
% \usepackage[pdfstringdefDisableCommands]{hyperref} % handle PDF string warnings
% \usepackage[english]{babel} % uncomment to change language

%% ---------------------------------------------------------------------
%%  2b. FIX FOR A HARMLESS HYPERREF WARNING (safe to ignore, but this
%%      silences it). Beamer loads babel internally, and when it builds
%%      a PDF bookmark for things like "Theorem" or "Appendix", hyperref
%%      can't put babel's internal \translate command into that PDF
%%      string, so it deletes it and warns:
%%        "Token not allowed in a PDF string (Unicode): removing `\translate'"
%%      This line tells hyperref to use the plain word instead of
%%      silently dropping it. It does not change how your slides look.
%% ---------------------------------------------------------------------
\pdfstringdefDisableCommands{%
  \def\translate#1{#1}%
}

%% ---------------------------------------------------------------------
%%  3. COLORS  >>> EDIT <<<
%%     Colors are given as RGB numbers from 0 to 255.
%%     Tip: search "RGB color picker" online and copy the numbers.
%%     This palette (navy + muted gold) is a classic, low-clash pairing:
%%     both colors are dark enough to read as text, and neither fights
%%     the other for attention the way navy + bright orange does.
%% ---------------------------------------------------------------------
\definecolor{MainColor}{RGB}{21, 42, 74}     % deep navy (titles, footer)
\definecolor{AccentColor}{RGB}{255, 51, 51} 
\definecolor{GoodColor}{RGB}{46, 125, 90}    % muted teal-green ("example" blocks)
\definecolor{LightGrey}{RGB}{246, 245, 241}  % warm off-white (code background)

%% Colors for the TITLE BOX on the title slide (set separately from
%% MainColor above, so you can give the title box its own look).
%% \colorlet copies MainColor's value here, so both stay in sync if
%% you only edit MainColor above. To make the box a DIFFERENT color,
%% replace the line with e.g. \definecolor{TitleBoxColor}{RGB}{...}.
\definecolor{TitleBoxColor}{RGB}{17, 36, 80}
\colorlet{TitleBoxTextColor}{white}

%% ---------------------------------------------------------------------
%%  4. LOOK AND FEEL (you can leave this alone at first)
%% ---------------------------------------------------------------------
%% Theme: try \usetheme{Madrid}, {Berlin}, {Copenhagen}, {Warsaw}...
%% But note: other themes have their own header/footer and can
%% conflict with the custom footer below.
\usetheme{default}

\setbeamercolor{structure}{fg=MainColor}         % bullets, links, TOC
\setbeamercolor{title}{fg=MainColor}             % big title text
\setbeamercolor{frametitle}{fg=white, bg=MainColor}  % slide title bar
\setbeamercolor{alerted text}{fg=AccentColor}    % \alert{...} color
\setbeamercolor{block title}{fg=white, bg=MainColor}
\setbeamercolor{block body}{bg=MainColor!8}
\setbeamercolor{block title alerted}{fg=white, bg=AccentColor}
\setbeamercolor{block body alerted}{bg=AccentColor!10}
\setbeamercolor{block title example}{fg=white, bg=GoodColor}
\setbeamercolor{block body example}{bg=GoodColor!10}

%% Color of the title box on the title slide (uses the colors defined
%% in section 3 above). Change TitleBoxColor / TitleBoxTextColor there,
%% not here.
\setbeamercolor{titlebox}{fg=TitleBoxTextColor, bg=TitleBoxColor}

\setbeamertemplate{navigation symbols}{}          % hide tiny arrows bottom-right
\setbeamertemplate{itemize items}[circle]         % bullet style: circle, square, ball
\setbeamertemplate{blocks}[rounded][shadow=false] % block style
\setbeamertemplate{caption}[numbered]             % "Figure 1:" style captions
\setbeamertemplate{bibliography item}[text]       % text labels in references
\setbeamertemplate{section in toc}[sections numbered] % numbered outline

%% Slide title bar (a full-width colored strip at the top)
\setbeamertemplate{frametitle}{%
  \nointerlineskip
  \begin{beamercolorbox}[wd=\paperwidth, ht=4.2ex, dp=1.6ex, leftskip=1em]{frametitle}
    \usebeamerfont{frametitle}\insertframetitle
  \end{beamercolorbox}%
}

%% Code style (used by \begin{lstlisting})
\lstset{
  basicstyle=\ttfamily\footnotesize,
  backgroundcolor=\color{LightGrey},
  keywordstyle=\color{MainColor}\bfseries,
  commentstyle=\color{gray}\itshape,
  frame=single, rulecolor=\color{MainColor!40},
  showstringspaces=false
}

%% ---------------------------------------------------------------------
%%  5. HELPER MACROS FOR LOGOS AND IMAGES
%%     \LogoBox{file}{height}{placeholder text}
%%     \ImageBox{file}{width}{placeholder text}
%%     If the file is missing, a placeholder box is shown.
%%     Always write the file extension (.png, .jpg, .pdf).
%% ---------------------------------------------------------------------
\newcommand{\LogoBox}[3]{%
  \IfFileExists{#1}%
    {\includegraphics[height=#2]{#1}}%
    {\fbox{\parbox[c][#2][c]{2.6cm}{\centering\scriptsize #3}}}%
}
\newcommand{\ImageBox}[3]{%
  \IfFileExists{#1}%
    {\includegraphics[width=#2]{#1}}%
    {\fbox{\parbox[c][3cm][c]{#2}{\centering\small #3}}}%
}
